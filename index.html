  <script>

    // 1. 指定のゆらぎクラス（.yuragi-char）をテキスト文字へ自動適用する処理

    function applyYuragiEffect(elementId) {

      const el = document.getElementById(elementId);

      if (!el) return;

      const text = el.innerText;

      el.innerHTML = '';

      [...text].forEach((char, index) => {

        const span = document.createElement('span');

        span.className = 'yuragi-char';

        span.style.animationDelay = `${(index * 0.15) % 3}s`;

        span.innerText = char === ' ' ? '\u00A0' : char;

        el.appendChild(span);

      });

    }


    // 対象要素にゆらぎを適用

    applyYuragiEffect('yuragi-title');

    applyYuragiEffect('yuragi-sub');

    applyYuragiEffect('yuragi-h2-1');

    applyYuragiEffect('yuragi-h2-2');

    applyYuragiEffect('yuragi-h2-3');


    // 2. Three.js 水面（GPGPU風波紋）波シミュレーション

    const container = document.getElementById('webgl-container');

    const scene = new THREE.Scene();

    const camera = new THREE.PerspectiveCamera(45, window.innerWidth / window.innerHeight, 0.1, 1000);

    camera.position.set(0, -10, 12);

    camera.lookAt(0, 0, 0);


    const renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });

    renderer.setSize(window.innerWidth, window.innerHeight);

    renderer.setPixelRatio(window.devicePixelRatio);

    container.appendChild(renderer.domElement);


    // 水面ジオメトリ

    const geometry = new THREE.PlaneGeometry(30, 20, 64, 64);

    const material = new THREE.MeshPhongMaterial({

      color: 0x1b4965,

      emissive: 0x0b2545,

      specular: 0x62b6cb,

      shininess: 80,

      wireframe: false,

      flatShading: true

    });


    const waterMesh = new THREE.Mesh(geometry, material);

    scene.add(waterMesh);


    // ライティング

    const ambientLight = new THREE.AmbientLight(0xffffff, 0.6);

    scene.add(ambientLight);


    const dirLight = new THREE.DirectionalLight(0xbee9e8, 1.2);

    dirLight.position.set(5, 10, 15);

    scene.add(dirLight);


    // 波のアニメーション計算

    let clock = new THREE.Clock();

    function animate() {

      requestAnimationFrame(animate);

      const t = clock.getElapsedTime() * 1.5;

      const pos = geometry.attributes.position;


      for (let i = 0; i < pos.count; i++) {

        const u = pos.getX(i);

        const v = pos.getY(i);

        const z = Math.sin(u * 0.5 + t) * 0.4 + Math.cos(v * 0.5 + t * 0.8) * 0.4;

        pos.setZ(i, z);

      }

      pos.needsUpdate = true;

      renderer.render(scene, camera);

    }

    animate();


    // 画面リサイズ対応

    window.addEventListener('resize', () => {

      camera.aspect = window.innerWidth / window.innerHeight;

      camera.updateProjectionMatrix();

      renderer.setSize(window.innerWidth, window.innerHeight);

    });
